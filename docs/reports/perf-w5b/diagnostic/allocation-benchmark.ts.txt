import { balancedCrowdModel } from '../../src/data/crowdAppearance';
import { l1Pedestrians } from '../../src/data/npcs';
import { keepsLook } from '../../src/sim/outbreak/appearance';
import type { EntitySnapshot } from '../../src/sim/world/types';
import type { SimWorld } from '../../src/sim/world/SimWorld';
import { Rng } from '../../src/core/Rng';
import type { Point } from '../../src/sim/npc/types';
import type { Refuge } from '../../src/sim/outbreak/Outbreak';
import { loadL4, runL4 } from '../../tools/sim-runner/l4Bots';

type Original = { world: SimWorld; rng: Rng; crowdModel(at: Point, infected?: boolean): string; refugeSpot(e: EntitySnapshot, refuge: Refuge): Point };


import { writeFileSync } from 'node:fs';
const { world, mission } = await loadL4(2, 'low');
try {
  runL4(world, mission, 'complete', { seed: 2, god: true, maxSeconds: 180, stopWhen: () => mission.state.l4!.turnAt > 0 && world.tick >= mission.state.l4!.turnAt + 10 });
  const current = world.npcs!.civilians.outbreak! as unknown as Original;
  const original = { world, rng: new Rng(1, 'allocation-benchmark') } as Original;
  Object.assign(current, { rng: new Rng(1, 'allocation-benchmark') });
original.crowdModel = function (at, infected = false) {
          return balancedCrowdModel(l1Pedestrians.models,
            [...this.world.entities.iterate()].filter(e => !e.hidden && e.health.current > 0 && Boolean(e.infected) === infected && keepsLook(e)).map(e => ({ model: e.appearance!.asset, x: e.transform.x, z: e.transform.z })), at, Math.floor(this.rng.next() * 0x100000000));
        };
original.refugeSpot = function (e, refuge) {
          const nav = this.world.infected!.nav, angle = Math.atan2(e.transform.z - refuge.z, e.transform.x - refuge.x);
          for (let radius = 1.4; radius <= 4; radius += .65) for (let k = 0; k < 9; k++) {
            const a = angle + ((k + e.id % 9) % 9 - 4) * Math.PI / 12;
            const p = { x: refuge.x + Math.cos(a) * radius, z: refuge.z + Math.sin(a) * radius };
            if (!nav.clear(p.x, p.z, .5) || !nav.visible(refuge, p, .35)) continue;
            let occupied = false;
            for (const other of this.world.entities.iterate()) if (other.id !== e.id && other.civilian && !other.hidden && other.health.current > 0) {
              const at = other.civilian.l1?.target ?? other.transform;
              if (Math.hypot(at.x - p.x, at.z - p.z) < .85) { occupied = true; break; }
            }
            if (!occupied) return p;
          }
          return { x: e.transform.x, z: e.transform.z };
        };
  const people = [...world.entities.iterate()].filter(e => e.civilian?.l1 && !e.hidden && e.health.current > 0).slice(0, 16);
  const refuge = world.npcs!.civilians.outbreak!.refuges.find(r => r.id.startsWith('refuge-door-'))!;
  for (const e of people) {
    if (current.crowdModel(e.transform) !== original.crowdModel(e.transform)) throw new Error('Model differs');
    if (JSON.stringify(current.refugeSpot(e, refuge)) !== JSON.stringify(original.refugeSpot(e, refuge))) throw new Error('Refuge differs');
  }
  const samples: { kind: string; name: string; ms: number }[] = [];
  for (const kind of ['model', 'refuge']) {
    const run = (subject: Original, repeats: number) => {
      const start = performance.now();
      for (let i = 0; i < repeats; i++) for (const e of people) {
        if (kind === 'model') subject.crowdModel(e.transform); else subject.refugeSpot(e, refuge);
      }
      return performance.now() - start;
    };
    run(original, 5); run(current, 5);
    for (const order of [['before','after'],['after','before'],['before','after'],['after','before']]) for (const name of order) samples.push({kind,name,ms:run(name === 'before' ? original : current, 100)});
  }
  const median = (kind: string, name: string) => { const values = samples.filter(s => s.kind === kind && s.name === name).map(s => s.ms).sort((a,b) => a-b); return (values[1]+values[2])/2; };
  const proof = { queriesPerSample: people.length * 100, equivalent: true, model: { before:median('model','before'),after:median('model','after') },refuge: { before:median('refuge','before'),after:median('refuge','after') },samples };
  writeFileSync('docs/reports/perf-w5b/diagnostic/allocation-benchmark.json',JSON.stringify(proof,null,2)); console.log(JSON.stringify(proof));
} finally { world.dispose(); }
