python3 - <<'PY'
import json,sys
from pathlib import Path
p=Path('test-results/ac-close/corrected-route-v7-results.json')
def passed(suites):
 for suite in suites:
  for spec in suite.get('specs',[]):
   if 'T-E24-17' in spec['title']:
    return all(t['status']=='expected' and any(r['status']=='passed' for r in t['results']) for t in spec['tests'])
  if passed(suite.get('suites',[])):return True
 return False
sys.exit(0 if p.exists() and passed(json.loads(p.read_text())['suites']) else 1)
PY

  rm -f /tmp/ac-close-l4-l6-verifier/primary-proof-hold /tmp/ac-close-l4-l6-verifier/priority-receipt
  exit "$proof_code"
