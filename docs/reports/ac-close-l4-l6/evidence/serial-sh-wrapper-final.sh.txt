#!/bin/sh
set -u
if [ "${MI_AC_BROWSER_HELD:-}" = 1 ] || [ "${MI_AC_NATIVE_TRY:-}" = 1 ]; then exec /bin/sh "$@"; fi
if [ "$1" = tools/perf/quiet-run.sh ]; then
  while [ -f /tmp/ac-close-l4-l6-verifier/defer-perf ]; do sleep 2; done
fi
case "$1" in
  tools/e2e-lock.sh)
    case " $* " in *playwright*) exec /bin/sh /tmp/ac-close-l4-l6-verifier/native-admit.sh "$@";; esac;;
  tools/perf/quiet-run.sh)
    case " $* " in
      *playwright*)
        export MI_AC_BROWSER_HELD=1
        exec /usr/bin/lockf -k /tmp/ac-close-l4-l6-browser.serial /bin/sh /tmp/ac-close-l4-l6-verifier/launch.sh "$@";;
    esac;;
  tools/remote/run.sh)
    case " $* " in
      *'npx playwright test'*|*'--pull dist'*)
        while [ -f /tmp/ac-close-l4-l6-verifier/primary-proof-hold ]; do sleep 2; done
        export MI_AC_BROWSER_HELD=1
        exec /usr/bin/lockf -k /tmp/ac-close-l4-l6-browser.serial /bin/sh /tmp/ac-close-l4-l6-verifier/launch.sh "$@";;
    esac;;
esac
exec /bin/sh "$@"
