  { type: 'combat.hit', tick: 1234, sourceId: 1, targetId: 77, attackId: 4, actionId: 'weapon.army-mg', amount: 14, damageType: 'bullet', position: { x: 1.5, y: .9, z: -3.25 }, direction: { x: .7, z: .7 }, critical: false },
  { type: 'ally.attack', tick: 1234, sourceId: 12, targetId: 90, role: 'soldier', from: { x: 1, y: 1.4, z: 2 }, to: { x: 5, y: 1, z: 7 } },

for (const [name, f] of [['structuredClone', structuredClone], ['cloneData', cloneData]] as const) {
  for (let r = 0; r < 3; r++) {
    const t = performance.now(); let n = 0;
    for (let i = 0; i < 200_000; i++) n += (f(events[i % 3]) as { tick: number }).tick;
    console.log(name, ((performance.now() - t) / 200_000 * 1e3).toFixed(3), 'us/event', n > 0);
  }
}
