"""Top self-time source lines from positionTicks. usage: lines.py profile bundle.js [top]"""
import json, sys, collections
p = json.load(open(sys.argv[1])); src = open(sys.argv[2]).read().split('\n'); top = int(sys.argv[3]) if len(sys.argv) > 3 else 40
interval = (p['endTime'] - p['startTime']) / max(1, len(p['samples']))
hits = collections.Counter(); fn = {}
for n in p['nodes']:
    for pt in n.get('positionTicks', []):
        hits[pt['line']] += pt['ticks']; fn[pt['line']] = n['callFrame']['functionName']
total = len(p['samples'])
print(f'samples {total}, {interval:.0f} us each')
# find enclosing file comment "// src/..." above line
markers = [(i + 1, l[3:].strip()) for i, l in enumerate(src) if l.startswith('// src/') or l.startswith('// tests/') or l.startswith('// node_modules/')]
import bisect
starts = [m[0] for m in markers]
for line, t in hits.most_common(top):
    k = bisect.bisect_right(starts, line) - 1; f = markers[k][1] if k >= 0 else '?'
    print(f'{100 * t / total:5.1f}% {line:6d} {f.split("/")[-1]:28s} {fn[line][:22]:22s} {src[line - 1].strip()[:150]}')
