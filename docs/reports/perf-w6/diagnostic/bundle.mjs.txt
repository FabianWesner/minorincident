// perf-w6 diagnostic: bundle the sim harness. With <origDir>, any source file that has a copy under origDir/<repo path>
// is read from that copy (the baseline revision), so base/new bundles differ only in the changed files.
// Usage: node tools/perf/w6/bundle.mjs <out.mjs> [origDir]
import { build } from 'esbuild';
import { existsSync, readFileSync } from 'node:fs';
import { relative, resolve } from 'node:path';
const [out, orig] = process.argv.slice(2);
const root = resolve(import.meta.dirname, '../../..');
const swap = orig ? [{ name: 'baseline', setup(b) {
  b.onLoad({ filter: /\/src\/.*\.ts$/ }, args => {
    const copy = resolve(orig, relative(root, args.path));
    return existsSync(copy) ? { contents: readFileSync(copy, 'utf8'), loader: 'ts' } : undefined;
  });
} }] : [];
await build({ entryPoints: [resolve(root, 'tools/perf/w6/sim-profile.ts')], bundle: true, platform: 'node', format: 'esm', outfile: out, external: ['@dimforge/rapier3d-compat'], logLevel: 'warning', plugins: swap,
  banner: { js: "import { createRequire } from 'module'; const require = createRequire(import.meta.url);" } });
