  objects++;
  const proto = Object.getPrototypeOf(v), kind = Array.isArray(v) ? 'Array' : proto === Object.prototype ? 'Object' : proto?.constructor?.name ?? 'null-proto';
  kinds.set(kind, (kinds.get(kind) ?? 0) + 1);
  if (seen.has(v)) { if (shared.length < 10) shared.push(`${path} = ${seen.get(v)}`); return; }

  else if (kind === 'Object') for (const k of Object.keys(v)) walk((v as Record<string, unknown>)[k], `${path}.${k}`);
};

const variant = (identity: boolean, arrayKeys: boolean) => (value: unknown) => {
  const seen = new Map<object, unknown>();

    if (v === null || typeof v !== 'object') return v;
    if (identity) { const known = seen.get(v); if (known !== undefined) return known; }

      const n = v.length, out: unknown[] = new Array(n); if (identity) seen.set(v, out);
      if (arrayKeys && Object.keys(v).length !== n) throw new Error('x');
      for (let i = 0; i < n; i++) out[i] = copy(v[i]);
      return out;
    }
    if (Object.getPrototypeOf(v) !== Object.prototype) throw new Error('y');
